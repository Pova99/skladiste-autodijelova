PART* load_parts(FILE* fp) {

	int count = 0;

	rewind(fp);

	if (fread(&count, sizeof(int), 1, fp) != 1) {

		return NULL;
	}

	if (count <= 0) {

		return NULL;
	}

	PART* p = calloc(count, sizeof(PART));

	if (!p) {

		perror("Calloc failed!");
		exit(EXIT_FAILURE);
	}

	size_t read = fread(p, sizeof(PART), count, fp);

	if (read != (size_t)count) {

		free(p);
		return NULL;
	}

	parts_num = count;

	return p;
}
